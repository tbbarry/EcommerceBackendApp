2. CRÉER LE SERVICE (src/app/services/attribute.service.ts)

import { Injectable } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { Observable } from 'rxjs';
import { AttributeWithValues } from '../models/attribute.model';

@Injectable({
  providedIn: 'root'
})
export class AttributeService {

  private apiUrl = 'http://localhost:8080/api/attributes';

  constructor(private http: HttpClient) { }

  /**
   * Récupérer tous les attributs avec leurs valeurs
   * Utilisé principalement pour le formulaire d'ajout de produit
   */
  getAllAttributesWithValues(): Observable<AttributeWithValues[]> {
    return this.http.get<AttributeWithValues[]>(`${this.apiUrl}/with-values`);
  }
}

3. UTILISER DANS LE COMPOSANT (product-add.component.ts)

import { Component, OnInit } from '@angular/core';
import { AttributeService } from './services/attribute.service';
import { AttributeWithValues } from './models/attribute.model';

@Component({
  selector: 'app-product-add',
  templateUrl: './product-add.component.html',
  styleUrls: ['./product-add.component.css']
})
export class ProductAddComponent implements OnInit {

  attributes: AttributeWithValues[] = [];
  loading = false;
  error: string | null = null;

  constructor(private attributeService: AttributeService) { }

  ngOnInit() {
    this.loadAttributes();
  }

  loadAttributes() {
    this.loading = true;
    this.error = null;

    this.attributeService.getAllAttributesWithValues().subscribe(
      (data) => {
        this.attributes = data;
        console.log('Attributs chargés:', this.attributes);
        this.loading = false;
      },
      (error) => {
        this.error = 'Erreur lors du chargement des attributs';
        console.error(error);
        this.loading = false;
      }
    );
  }
}

5. EXEMPLE AVANCÉ - CRÉATION DE VARIANTES

export class ProductAddComponent implements OnInit {

  attributes: AttributeWithValues[] = [];
  selectedAttributes: Map<number, number[]> = new Map(); // attributeId -> [valueIds]
  variants: Variant[] = [];

  // ... autres propriétés

  ngOnInit() {
    this.loadAttributes();
  }

  /**
   * Appelée quand l'utilisateur coche une valeur d'attribut
   */
  onAttributeValueSelected(attributeId: number, valueId: number, event: any) {
    if (!this.selectedAttributes.has(attributeId)) {
      this.selectedAttributes.set(attributeId, []);
    }

    const values = this.selectedAttributes.get(attributeId)!;
    
    if (event.target.checked) {
      values.push(valueId);
    } else {
      const index = values.indexOf(valueId);
      if (index > -1) {
        values.splice(index, 1);
      }
    }

    console.log('Attributs sélectionnés:', this.selectedAttributes);
  }

  /**
   * Créer une variante avec les attributs sélectionnés
   */
  createVariantWithSelectedAttributes() {
    // Vérifier qu'au moins un attribut est sélectionné par type
    if (this.selectedAttributes.size === 0) {
      alert('Veuillez sélectionner au moins une valeur pour chaque attribut');
      return;
    }

    // Construire la variante
    const attributeValues = Array.from(this.selectedAttributes.entries()).map(([attributeId, valueIds]) => {
      return valueIds.map(valueId => ({
        attributeValueId: valueId
      }));
    }).flat();

    const variant: Variant = {
      sku: this.generateSku(),
      price: this.form.get('price')?.value,
      quantity: this.form.get('quantity')?.value,
      reservedQuantity: 0,
      attributeValues: attributeValues
    };

    this.variants.push(variant);
    this.selectedAttributes.clear(); // Réinitialiser pour la prochaine variante
    console.log('Variante créée:', variant);
  }

  /**
   * Afficher les variantes créées
   */
  displayVariants() {
    return this.variants.map(v => {
      const attributesText = v.attributeValues
        .map(av => this.getAttributeValueName(av.attributeValueId))
        .join(' | ');
      
      return `${v.sku} - ${attributesText} - €${v.price}`;
    });
  }

  /**
   * Récupérer le nom d'une valeur d'attribut par son ID
   */
  getAttributeValueName(valueId: number): string {
    for (const attr of this.attributes) {
      const value = attr.values.find(v => v.id === valueId);
      if (value) {
        return `${attr.name}: ${value.value}`;
      }
    }
    return 'Valeur inconnue';
  }

  private generateSku(): string {
    // Exemple: PROD-001, PROD-002, etc.
    return `PROD-${String(this.variants.length + 1).padStart(3, '0')}`;
  }
}

🔗 ENDPOINTS COMPLÉMENTAIRES

GET /api/attributes
  → Récupère tous les attributs (sans optimisation spéciale)

GET /api/attributes/{id}
  → Récupère un attribut spécifique par ID

GET /api/attribute-values
  → Récupère toutes les valeurs d'attribut

GET /api/attribute-values/by-attribute/{attributeId}
  → Récupère les valeurs d'un attribut spécifique

⚠️ POINTS IMPORTANTS

✓ L'endpoint /api/attributes/with-values est OPTIMISÉ pour le front
✓ Il retourne TOUT en une seule requête (pas de N+1 queries)
✓ AUCUNE authentification requise (endpoint public)
✓ Les données sont hiérarchisées (attribut → valeurs)
✓ À charger UNE SEULE FOIS au démarrage du formulaire
