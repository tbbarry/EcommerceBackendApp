🏗️ COMPOSANT PRINCIPAL (product-add.component.ts)

export class ProductAddComponent implements OnInit {

  form: FormGroup;
  
  // Données statiques chargées au démarrage
  attributes: AttributeDto[] = [];
  categories: CategoryDto2[] = [];
  
  // Données du formulaire
  variants: VariantFullRequest[] = [];
  images: ProductImageRequest[] = [];
  
  constructor(
    private formBuilder: FormBuilder,
    private productService: ProductService,
    private attributeService: AttributeService,
    private categoryService: CategoryService,
    private snackBar: MatSnackBar,
    private router: Router
  ) {}
  
  ngOnInit() {
    this.initForm();
    this.loadStaticData();
  }
  
  private initForm() {
    this.form = this.formBuilder.group({
      // Onglet 1: Infos de base
      name: ['', [Validators.required, Validators.minLength(3)]],
      description: ['', [Validators.required]],
      price: [0, [Validators.required, Validators.min(0)]],
      brand: ['', Validators.required],
      categoryId: ['', Validators.required],
      slug: [''] // Auto-généré mais éditable
    });
  }
  
  private loadStaticData() {
    // Charger attributs (avec toutes leurs valeurs)
    this.attributeService.getAll().subscribe(
      (attrs) => this.attributes = attrs,
      (error) => this.showError('Erreur chargement attributs')
    );
    
    // Charger catégories
    this.categoryService.getAll().subscribe(
      (cats) => this.categories = cats,
      (error) => this.showError('Erreur chargement catégories')
    );
  }
  
  // Méthodes pour gérer les variantes
  addVariant(formData: any) {
    const variant: VariantFullRequest = {
      sku: formData.sku, // Généré ou fourni
      price: formData.price,
      quantity: formData.quantity,
      reservedQuantity: formData.reservedQuantity || 0,
      attributeValues: formData.selectedAttributeValues // List des IDs sélectionnés
    };
    
    // Vérifier qu'on n'a pas la même combo d'attributs
    if (this.isDuplicateVariant(variant)) {
      this.showError('Cette combinaison de variante existe déjà');
      return;
    }
    
    this.variants.push(variant);
  }
  
  removeVariant(index: number) {
    this.variants.splice(index, 1);
  }
  
  isDuplicateVariant(newVariant: VariantFullRequest): boolean {
    // Comparer les IDs des attributs sélectionnés
    return this.variants.some(v => 
      this.hasSameAttributes(v.attributeValues, newVariant.attributeValues)
    );
  }
  
  hasSameAttributes(attrs1: VariantAttributeValueRequest[], attrs2: VariantAttributeValueRequest[]): boolean {
    if (attrs1.length !== attrs2.length) return false;
    return attrs1.every(a1 => attrs2.some(a2 => a2.attributeValueId === a1.attributeValueId));
  }
  
  // Méthodes pour gérer les images
  addImage(imageRequest: ProductImageRequest) {
    this.images.push(imageRequest);
  }
  
  removeImage(index: number) {
    this.images.splice(index, 1);
  }
  
  setMainImage(index: number) {
    this.images.forEach((img, i) => {
      img.isMain = (i === index);
    });
  }
  
  reorderImages(images: ProductImageRequest[]) {
    this.images = images;
    this.images.forEach((img, i) => {
      img.displayOrder = i + 1;
    });
  }
  
  // Auto-générer le slug
  generateSlug() {
    const name = this.form.get('name')?.value;
    if (name) {
      const slug = name
        .toLowerCase()
        .replace(/[àâä]/g, 'a')
        .replace(/[éèêë]/g, 'e')
        .replace(/[ôö]/g, 'o')
        .replace(/[ù]/g, 'u')
        .replace(/\s+/g, '-')
        .replace(/[^a-z0-9-]/g, '')
        .replace(/-+/g, '-')
        .replace(/^-|-$/g, '');
      this.form.patchValue({ slug: slug });
    }
  }
  
  // Validation globale
  isFormValid(): boolean {
    return this.form.valid && 
           this.variants.length > 0 && 
           this.images.length > 0;
  }
  
  // Publier le produit
  publishProduct() {
    if (!this.isFormValid()) {
      this.showError('Veuillez remplir tous les champs obligatoires');
      return;
    }
    
    const request: ProductCreateRequest = {
      ...this.form.value,
      variants: this.variants,
      images: this.images
    };
    
    this.productService.createFull(request).subscribe(
      (response) => {
        this.showSuccess('✅ Produit créé avec succès !');
        this.router.navigate(['/products', response.id]);
      },
      (error) => {
        this.showError('❌ Erreur : ' + error.message);
      }
    );
  }
  
  private showError(message: string) {
    this.snackBar.open(message, 'Fermer', { duration: 5000, panelClass: 'error-snackbar' });
  }
  
  private showSuccess(message: string) {
    this.snackBar.open(message, 'Fermer', { duration: 3000, panelClass: 'success-snackbar' });
  }
}
